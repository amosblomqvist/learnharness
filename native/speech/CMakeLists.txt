cmake_minimum_required(VERSION 3.20)
project(learnharness_speech LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PIPER_ROOT "" CACHE PATH "Installed pinned Piper v1.8.0 tree")
find_path(PIPER_INCLUDE piper.h PATHS "${PIPER_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
find_library(PIPER_LIBRARY NAMES piper libpiper PATHS "${PIPER_ROOT}/lib" "${PIPER_ROOT}/bin" NO_DEFAULT_PATH REQUIRED)
add_executable(piper-helper piper-worker.cpp)
target_include_directories(piper-helper PRIVATE "${PIPER_INCLUDE}")
target_link_libraries(piper-helper PRIVATE "${PIPER_LIBRARY}")
set_target_properties(piper-helper PROPERTIES INSTALL_RPATH "$ORIGIN/lib;@loader_path/lib")
install(TARGETS piper-helper RUNTIME DESTINATION .)

if(UNIX AND NOT APPLE)
  target_link_options(piper-helper PRIVATE "-Wl,--disable-new-dtags")
endif()
